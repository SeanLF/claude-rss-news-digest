# Production dependency pins (applied via `uv pip install -c` in newsroom/Dockerfile).
#
# Why: pyproject.toml lists claude-agent-sdk unpinned, and both Dockerfiles install
# with `uv pip install -r pyproject.toml` (the uv.lock is not consulted), so without
# this a fresh PROD build floats to whatever PyPI-latest is at build time -- a bad SDK
# release would break prod with no guard. Pinning here makes prod builds reproducible.
#
# CI (newsroom/Dockerfile.ci) deliberately does NOT apply this constraint: it floats to
# latest so a bad release fails the CI build first, not prod. The canary in
# newsroom/tests/test_sdk_pin.py ties this pin to the SDK version the in-repo SDK
# workarounds (#378 teardown timeout, _thinking_for thinking-400) were validated against,
# so bumping the pin forces re-verifying them.
claude-agent-sdk==0.2.143

# Scientific stack (used by cluster_extractjoin's TF-IDF/agglomerative join). These are
# COMPILED wheels, and the runtime base is a compiler-less python:3.14-slim -- so a future
# rebuild that resolved a version WITHOUT a cp314 manylinux wheel would fall back to a source
# build and FAIL the prod build hard. Pinning to the wheel-backed versions verified on the
# slim/amd64 target keeps prod builds reproducible and wheel-installable. Bump deliberately
# after confirming the new versions ship cp314 manylinux wheels.
numpy==2.5.2
scipy==1.18.0
scikit-learn==1.9.0

# trafilatura (fulltext.py's full-text fetch for SELECTED stories). Same compiler-less
# python:3.14-slim constraint as the scientific stack above: trafilatura pulls in lxml, a
# COMPILED wheel, and its other deps (htmldate, courlan, justext, ...) are pure Python. Pinning
# both keeps the prod build wheel-installable and reproducible -- verified empirically by
# building newsroom/Dockerfile on this exact pin (cp314 manylinux wheels, no source build).
trafilatura==2.2.0
lxml==6.1.2
